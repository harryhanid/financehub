```python
# modules/general_ledger/coa_map.py
"""General Ledger MR: account -> COA HO / category mapping from the 'COA' sheet of
Management Report - MASTER.xlsb."""
import hashlib
import json
import re
from datetime import datetime

from pyxlsb import open_workbook

from database import get_conn

SHEET_NAME = "COA"
_HEADER_ROW = 1                                   # 0-based; row 0 is a numbering row
_SOURCE_COL = {"SMT": 1, "ETF": 3}                # COA SMT, COA ETF
_COA_HO_COL = 6
_CATEGORY_COL = 9                                 # CATEGORIES 2, already shaped like ' 10 - Office ... '
_EXPECTED_HEADER = {1: "COA SMT", 3: "COA ETF", 6: "COA HO", 9: "CATEGORIES 2"}
_ERR = re.compile(r"^0x[0-9A-Fa-f]{1,2}$")        # how pyxlsb shows an Excel error cell


class CoaParseError(ValueError):
    """Raised with a user-facing message when the COA sheet cannot be used."""


def _text(v) -> str:
    if v is None:
        return ""
    if isinstance(v, float) and v.is_integer():
        return str(int(v))
    s = str(v).strip()
    return "" if _ERR.match(s) else s


def _coa_ho(v):
    if isinstance(v, (int, float)):
        return str(int(v)) if v else None
    s = _text(v)
    return s if s.isdigit() else None


def _cell(row, i):
    return row[i] if i < len(row) else None


def normalise_coa(sheet_rows) -> list:
    if len(sheet_rows) <= _HEADER_ROW:
        raise CoaParseError(f"Sheet '{SHEET_NAME}' kosong.")
    header = sheet_rows[_HEADER_ROW]
    for pos, label in _EXPECTED_HEADER.items():
        actual = " ".join(str(_cell(header, pos) or "").split()).lower()
        if actual != label.lower():
            raise CoaParseError(
                f"Header sheet '{SHEET_NAME}' tidak sesuai: kolom {pos + 1} harus '{label}'.")
    rows, seen = [], set()
    for r in sheet_rows[_HEADER_ROW + 1:]:
        coa_ho = _coa_ho(_cell(r, _COA_HO_COL))
        if coa_ho is None:
            continue
        category = _text(_cell(r, _CATEGORY_COL))
        for source, col in _SOURCE_COL.items():
            account = _text(_cell(r, col))
            if account and (source, account) not in seen:
                seen.add((source, account))
                rows.append({"source": source, "account": account,
                             "coa_ho": coa_ho, "category": category})
    return rows


def read_coa_sheet(fileobj) -> list:
    try:
        wb = open_workbook(fileobj)
    except Exception as e:
        raise CoaParseError(f"File tidak bisa dibaca sebagai .xlsb: {e}") from e
    try:
        with wb:
            if SHEET_NAME not in wb.sheets:
                raise CoaParseError(f"Sheet '{SHEET_NAME}' tidak ditemukan di file.")
            with wb.get_sheet(SHEET_NAME) as sh:
                return normalise_coa([[c.v for c in r] for r in sh.rows()])
    except CoaParseError:
        raise
    except Exception as e:
        raise CoaParseError(f"File tidak bisa dibaca: {e}") from e


_DDL = """
CREATE TABLE IF NOT EXISTS gl_coa_map (
    source   TEXT NOT NULL,
    account  TEXT NOT NULL,
    coa_ho   TEXT NOT NULL,
    category TEXT NOT NULL DEFAULT '',
    PRIMARY KEY (source, account)
);
CREATE TABLE IF NOT EXISTS gl_coa_sync_log (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    synced_at    TEXT,
    synced_by    TEXT,
    filename     TEXT,
    content_hash TEXT,
    rows_smt     INTEGER DEFAULT 0,
    rows_etf     INTEGER DEFAULT 0
);
"""


def ensure_schema(conn) -> None:
    conn.executescript(_DDL)
    conn.commit()


def compute_hash(rows) -> str:
    lines = sorted(json.dumps([r["source"], r["account"], r["coa_ho"], r["category"]],
                              ensure_ascii=False) for r in rows)
    h = hashlib.sha256()
    for line in lines:
        h.update(line.encode("utf-8"))
        h.update(b"\n")
    return h.hexdigest()


def get_map() -> dict:
    conn = get_conn()
    try:
        ensure_schema(conn)
        return {(r["source"], r["account"]): (r["coa_ho"], r["category"])
                for r in conn.execute("SELECT source, account, coa_ho, category FROM gl_coa_map")}
    finally:
        conn.close()


def last_sync():
    conn = get_conn()
    try:
        ensure_schema(conn)
        row = conn.execute("SELECT * FROM gl_coa_sync_log ORDER BY id DESC LIMIT 1").fetchone()
        return dict(row) if row else None
    finally:
        conn.close()


def _count_rows() -> int:
    conn = get_conn()
    try:
        ensure_schema(conn)
        return conn.execute("SELECT COUNT(*) FROM gl_coa_map").fetchone()[0]
    finally:
        conn.close()


def sync_coa(filename, user, rows) -> dict:
    """Replace the mapping with `rows` unless identical to the last sync (and still in the table)."""
    content_hash = compute_hash(rows)
    n_smt = sum(1 for r in rows if r["source"] == "SMT")
    n_etf = len(rows) - n_smt
    last = last_sync()
    if (last and last["content_hash"] == content_hash
            and _count_rows() == last["rows_smt"] + last["rows_etf"]):
        return {"changed": False, "rows_smt": n_smt, "rows_etf": n_etf}
    conn = get_conn()
    try:
        ensure_schema(conn)
        try:
            conn.execute("DELETE FROM gl_coa_map")
            conn.executemany(
                "INSERT INTO gl_coa_map (source, account, coa_ho, category) VALUES (?,?,?,?)",
                [(r["source"], r["account"], r["coa_ho"], r["category"]) for r in rows])
            conn.execute(
                "INSERT INTO gl_coa_sync_log (synced_at, synced_by, filename, content_hash, "
                "rows_smt, rows_etf) VALUES (?,?,?,?,?,?)",
                (datetime.now().isoformat(timespec="seconds"), user, filename, content_hash,
                 n_smt, n_etf))
            conn.commit()
        except Exception:
            conn.rollback()
            raise
    finally:
        conn.close()
    return {"changed": True, "rows_smt": n_smt, "rows_etf": n_etf}
```

```python
# modules/general_ledger/mr_rows.py
"""General Ledger MR: build Management Report (Append1 format) rows from GL SMT / GL ETF rows.

Mirrors the 'TransferDataGLCombineFinalFix' macro and the GL Combine sheet of
Management Report - MASTER.xlsb. Pure functions: no database access.
"""
from collections import namedtuple
from datetime import date
from decimal import Decimal, ROUND_HALF_UP

UNMAPPED = "(Belum dipetakan)"
_MIO = Decimal(1000000)
_TENTH = Decimal("0.1")
_ELIGIBLE_PREFIX = "6789"

MrBuild = namedtuple("MrBuild", "rows skipped_date")


def excel_round_mio(amount) -> float:
    """(Debit - Credit) in rupiah -> Rp Mio, 1 decimal, ties away from zero like Excel's ROUND."""
    d = Decimal(str(float(amount))) / _MIO
    return float(d.quantize(_TENTH, rounding=ROUND_HALF_UP)) + 0.0      # + 0.0 turns -0.0 into 0.0


def _parse_date(value):
    try:
        return date.fromisoformat(str(value)[:10])
    except ValueError:
        return None


def build_mr_rows(source_rows: dict, coa_map: dict) -> MrBuild:
    rows, skipped = [], 0
    for source in ("SMT", "ETF"):
        for r in source_rows.get(source, []):
            account = str(r.get("acc_management") or "").strip()
            amount = r.get("debit_credit") or 0.0
            if not account or account[0] not in _ELIGIBLE_PREFIX or not amount:
                continue
            d = _parse_date(r.get("tgl"))
            if d is None:
                skipped += 1
                continue
            coa, category = coa_map.get((source, account), ("", ""))
            memo = r.get("allocation_memo") or ""
            rows.append({
                "coa": coa, "mm": d.month, "yy": d.year,
                "cc": str(r.get("job_name") or "").strip(),
                "category": category or UNMAPPED,
                "activity": str(r.get("budget_activity") or "").strip(),
                "description": f"- Act {d.year} : {memo} Month : {d.month}",
                "custom": "3", "mm1": d.month, "pta": 0.0, "bgt": 0.0,
                "act": excel_round_mio(amount), "source": source, "account": account,
                "unmapped": not category,
            })
    return MrBuild(rows, skipped)


def unmapped_summary(rows) -> list:
    out = {}
    for r in rows:
        if r["unmapped"]:
            e = out.setdefault((r["source"], r["account"]),
                               {"source": r["source"], "account": r["account"], "rows": 0, "act": 0.0})
            e["rows"] += 1
            e["act"] = round(e["act"] + r["act"], 6)
    return sorted(out.values(), key=lambda e: (e["source"], e["account"]))


def period_options(rows) -> dict:
    months = {}
    for r in rows:
        months.setdefault(r["yy"], set()).add(r["mm"])
    return {"yys": sorted(months), "months": {y: sorted(m) for y, m in months.items()}}


def apply_cutoff(rows, yy, mm) -> list:
    """Cumulative cutoff: the selected year, January up to and including month `mm`."""
    return [r for r in rows if r["yy"] == yy and r["mm"] <= mm]
```

```python
# modules/general_ledger/mr_report.py
"""General Ledger MR: group MR rows into the report layout and the flat data table.

Pure functions over the row dicts produced by mr_rows.build_mr_rows.
"""
from modules.general_ledger.mr_rows import UNMAPPED


def list_ccs(rows) -> list:
    return sorted({r["cc"] for r in rows if r["cc"]})


def _category_order(category):
    return (category == UNMAPPED, category)           # unmapped always last


def build_report(rows, cc=None) -> dict:
    selected = [r for r in rows if not cc or r["cc"] == cc]
    cats = {}
    for r in selected:
        key = (r["custom"], r["mm1"], r["description"], r["activity"])
        lines = cats.setdefault(r["category"], {})
        lines[key] = lines.get(key, 0.0) + r["act"]
    groups = []
    for category in sorted(cats, key=_category_order):
        lines = [{"custom": k[0], "mm1": k[1], "description": k[2], "activity": k[3],
                  "act": round(v, 6)} for k, v in sorted(cats[category].items(),
                                                          key=lambda kv: (kv[0][1], kv[0][2], kv[0][3]))]
        groups.append({"category": category, "lines": lines,
                       "total": round(sum(l["act"] for l in lines), 6)})
    return {"has_data": bool(groups), "cc": cc, "groups": groups,
            "grand": round(sum(g["total"] for g in groups), 6)}


def query_data(rows, q=None, cc=None, page=1, per_page=100) -> dict:
    needle = (q or "").strip().lower()
    selected = [r for r in rows
                if (not cc or r["cc"] == cc)
                and (not needle or needle in r["description"].lower() or needle in r["cc"].lower()
                     or needle in r["activity"].lower() or needle in r["coa"].lower())]
    selected.sort(key=lambda r: (r["mm"], r["source"], r["cc"], r["description"]))
    total = len(selected)
    pages = max(1, -(-total // per_page))
    page = min(max(1, page), pages)
    return {"rows": selected[(page - 1) * per_page: page * per_page], "total": total,
            "sum_act": round(sum(r["act"] for r in selected), 6),
            "page": page, "pages": pages, "per_page": per_page}
```

```python
    """Rp Mio with 1 decimal, Indonesian style: 1234.56 -> '1.234,6'; None -> ''."""

    return f"{float(value):,.1f}".replace(",", "\0").replace(".", ",").replace("\0", ".")
```

```python
    try:
        return int(request.args.get(name))

def _resolve_period(opts):
    """Requested year/month if valid, otherwise the latest period that has data."""
    yy = _int_arg("yy")
    if yy not in opts["yys"]:
        yy = opts["yys"][-1] if opts["yys"] else None
    if yy is None:
        return None, None
    months = opts["months"][yy]
    mm = _int_arg("mm")
    return yy, (mm if mm in months else months[-1])

        return jsonify({"ok": False, "pesan": "File belum dipilih."}), 400
    if not f.filename.lower().endswith(".xlsb"):
        return jsonify({"ok": False, "pesan": "File harus berformat .xlsb."}), 400
    try:
        rows = coa_map.read_coa_sheet(io.BytesIO(f.read()))
        if not rows:
            raise coa_map.CoaParseError("Tidak ada pemetaan COA yang valid di sheet COA.")
    except coa_map.CoaParseError as e:
        return jsonify({"ok": False, "pesan": str(e)}), 400
    try:
        out = coa_map.sync_coa(f.filename, get_jwt().get("username", ""), rows)
    except Exception as e:
        return jsonify({"ok": False, "pesan": f"Sync gagal, pemetaan lama tidak berubah: {e}"}), 500
    if not out["changed"]:
        pesan = (f"Sudah sinkron: pemetaan COA tidak berubah "
                 f"(SMT {out['rows_smt']} akun, ETF {out['rows_etf']} akun).")
    else:
        pesan = f"Sync COA selesai: SMT {out['rows_smt']} akun, ETF {out['rows_etf']} akun."
    return jsonify({"ok": True, "pesan": pesan, **out})
```

```python
                                reason="GL - MASTER.xlsb / Management Report - MASTER harry.xlsb not reachable")


def _norm(text):
    return " ".join(str(text or "").split())


def _coa(v):
    return str(int(v)) if isinstance(v, (int, float)) else ""


@pytest.fixture(scope="module")
def files(tmp_path_factory):
    d = tmp_path_factory.mktemp("mrreal")
    gl, mr = str(d / "gl.xlsb"), str(d / "mr.xlsb")
    shutil.copy(GL, gl)                         # the masters are usually open in Excel
    shutil.copy(MR, mr)
    return gl, mr


@pytest.fixture(scope="module")
def built(files):
    gl, mr = files
    cmap = {(r["source"], r["account"]): (r["coa_ho"], r["category"])
            for r in coa_map.read_coa_sheet(mr)}
    source = {label: read_gl_sheet(gl, sheet).rows
              for label, sheet in (("SMT", "GL SMT"), ("ETF", "GL ETF"))}
    return build_mr_rows(source, cmap)


def ours_key(r):
    return (r["source"], r["coa"], r["mm"], r["cc"], r["category"], _norm(r["description"]), r["act"])


def test_rows_match_the_gl_combine_sheet(files, built):
    _, mr = files
    theirs = Counter()
    with open_workbook(mr) as wb, wb.get_sheet("GL Combine") as sh:
        for i, row in enumerate(sh.rows()):
            v = [c.v for c in row] + [None] * 17
            if i == 0 or v[7] not in ("SMT", "ETF"):
                continue
            theirs[(v[7], _coa(v[8]), int(v[10]), str(v[5] or "").strip(), str(v[16] or "").strip(),
                    _norm(v[15]), excel_round_mio(v[6]))] += 1
    ours = Counter(ours_key(r) for r in built.rows)
    assert not (ours - theirs), list((ours - theirs).items())[:5]
    assert not (theirs - ours), list((theirs - ours).items())[:5]
    assert built.skipped_date == 0
    assert not [r for r in built.rows if r["unmapped"]]


def test_our_act_values_exist_in_append1_with_the_same_rounding(files, built):
    _, mr = files
    append3 = Counter()
    with open_workbook(mr) as wb, wb.get_sheet("Append1") as sh:
        for i, row in enumerate(sh.rows()):
            v = [c.v for c in row] + [None] * 13
            if i == 0 or str(v[9]) != "3":
                continue
            append3[(_coa(v[0]), int(v[1]), str(v[3] or "").strip(), str(v[5] or "").strip(),
                     _norm(v[7]), round(float(v[11] or 0), 1))] += 1
    ours = Counter((r["coa"], r["mm"], r["cc"], r["category"], _norm(r["description"]), r["act"])
                   for r in built.rows)
    missing = ours - append3
    assert not missing, list(missing.items())[:5]
```
